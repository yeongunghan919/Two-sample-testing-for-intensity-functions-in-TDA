\tikzset{

  leftpoint/.style={
    circle, fill=tealpoint, draw=white,

  rightpoint/.style={
    circle, fill=bluepoint, draw=white,
    line width=0.55pt, inner sep=0pt, minimum size=7.5pt
  },
  axislabel/.style={font=\large\itshape, text=labelgray},
  paneltitle/.style={font=\Large},
}
